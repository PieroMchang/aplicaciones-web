src/shows/domain/model/episode.entity.js

JavaScript
/**
 * Domain entity representing a TV Show Episode.
 * @summary Encapsulates the core attributes of an episode as required by the TVMaze API.
 * @author Piero Francisco Montes Chang
 */
export class Episode {
    constructor({ id = 0, name = '', season = 0, number = 0, summary = '', url = '', image = null }) {
        this.id = id;
        this.name = name;
        this.season = season;
        this.number = number;
        // Limpiamos etiquetas HTML del resumen si la API las trae
        this.summary = summary ? summary.replace(/<\/?[^>]+(>|$)/g, "") : "No summary available.";
        this.url = url;
        // Obtenemos la url de la imagen desde image.medium
        this.image = image && image.medium ? image.medium : 'https://placehold.co/600x400?text=No+Image';
    }
}
src/shows/infrastructure/shows.service.js

JavaScript
import axios from "axios";

/**
 * Infrastructure adapter for TVMaze HTTP endpoints.
 * @summary Isolates external transport concerns from the application layers.
 * @author Piero Francisco Montes Chang
 */
const tvMazeApiUrl = import.meta.env.VITE_TVMAZE_API_URL;
const episodesEndpoint = import.meta.env.VITE_EPISODES_ENDPOINT_PATH;

const http = axios.create({
    baseURL: tvMazeApiUrl
});

export class ShowsService {
    /**
     * Retrieves episodes from the TVMaze API.
     * @returns {Promise<import('axios').AxiosResponse>}
     */
    getEpisodes = () => http.get(`${episodesEndpoint}`);
}
src/shows/infrastructure/episode.assembler.js

JavaScript
import { Episode } from "@/shows/domain/model/episode.entity.js";

/**
 * Infrastructure service that maps episode data from API responses into Domain Entities.
 * @summary Acts as a Data Mapper between the API JSON and the Episode domain entity.
 * @author Piero Francisco Montes Chang
 */
export class EpisodeAssembler {
    /**
     * Maps an array of API objects to Episode entities.
     * @param {Array} responseData - The raw data array from the Axios response.
     * @returns {Episode[]} An array of Episode domain entities.
     */
    toEntitiesFromResponse(responseData) {
        if (!Array.isArray(responseData)) return [];
        return responseData.map(episode => new Episode(episode));
    }
}

src/shows/presentation/components/episode-list.component.vue

Fragmento de código
<script setup>
import { ref, onMounted } from 'vue';
import { useI18n } from 'vue-i18n';
import EpisodeCard from './episode-card.component.vue';
import { ShowsService } from '@/shows/infrastructure/shows.service.js';
import { EpisodeAssembler } from '@/shows/infrastructure/episode.assembler.js';

/**
 * Presentation component for rendering the list of episodes.
 * @summary Fetches data from ShowsService and maps it to EpisodeCard components.
 * @author Piero Francisco Montes Chang
 */
const { t } = useI18n();
const episodes = ref([]);
const showsService = new ShowsService();
const episodeAssembler = new EpisodeAssembler();

onMounted(async () => {
  try {
    const response = await showsService.getEpisodes();
    episodes.value = episodeAssembler.toEntitiesFromResponse(response.data);
  } catch (error) {
    console.error("Error fetching episodes:", error);
  }
});
</script>

<template>
  <div class="p-4">
    <h2 class="text-center mb-4 text-3xl font-bold" aria-label="Section title">{{ t('view.title') }}</h2>
    <div class="grid">
      <div class="col-12 md:col-6 lg:col-4 xl:col-3" v-for="episode in episodes" :key="episode.id">
        <episode-card :episode="episode" />
      </div>
    </div>
  </div>
</template>
